### Dockerfile

- `FROM`: 어떤 OS나 환경에서 시작할 것인가? (예: openjdk, node, python)
- `WORKDIR`: 컨테이너 안에서 명령어를 실행할 '작업 폴더' 설정
- `COPY` / `ADD`: 로컬에 있는 소스 코드를 컨테이너 안으로 복사 
	- 예: Nginx 기본 설정 파일 대신 새로 만든 설정 파일(nginx.conf)로 덮어씌움.
- `RUN`: 이미지를 만드는 단계에서 실행할 명령어 (라이브러리 설치, 빌드 등)
- `EXPOSE`: 컨테이너가 사용할 포트 번호 명시
- `CMD`: 컨테이너가 딱 켜졌을 때 실행할 '최종 명령어' (서버 실행 등)

### docker-compose.yml

``` yml
# 공통 로그 설정 템플릿: 무제한 로그 적재로 인한 Disk Full 장애 방지 (최대 10MB짜리 파일 3개)
x-logging-default: &logging-default
  driver: "json-file"
  options:
    max-size: "10m"
    max-file: "3"

services:
	# 1. Frontend (Vue.js + Nginx)
	capture-frontend:
		image: ghcr.io/<깃허브 아이디>/<저장소 이름>:latest # GHCR에서 이미지를 가져옴
		restart: unless-stopped
		logging: *logging-default
		# 외부 포트 노출 제거 (Cloudflare Tunnel로만 통신)
		# ports:
		#   - "80:80"        # 단계 1에서는 HTTP(80) 1개만 열어둡니다. (추후 운영에서는 443 추가)
	    networks:
			- frontend-network
			- backend-network
	    depends_on:
			- capture-server
	    healthcheck:
			# 80포트가 열려있는지 확인. 나중에 https로 변경하면 이 부분도 변경해야함.
			test: ["CMD-SHELL", "wget -q -O /dev/null http://localhost/ || curl -sf http://localhost/ || exit 1"]
			interval: 60s
			timeout: 10s
			retries: 3
			start_period: 30s
		
	# 2. Database (MySQL - 운영 환경은 이 이미지 사용을 권장)
	db:
	    image: mysql:8.0
	    restart: unless-stopped
	    logging: *logging-default
	    networks:
	      - data-network
	    # e2-micro 메모리 최적화 옵션 적용
	    command:
	      - --performance_schema=OFF
	      - --innodb_buffer_pool_size=64M
	      - --max_connections=50
	    environment: 
	      MYSQL_ROOT_PASSWORD: ${MYSQL_ROOT_PASSWORD:?MYSQL_ROOT_PASSWORD is required}
	      MYSQL_DATABASE: ${DB_NAME:-capture_db}
	      MYSQL_USER: ${DB_USERNAME:-capture_app}
	      MYSQL_PASSWORD: ${DB_PASSWORD:?DB_PASSWORD is required}
	      TZ: Asia/Seoul
	    volumes:
			# 처음 빌드 시에 init.sql 실행 (데이터베이스, 테이블 생성 쿼리)
			- ./init.sql:/docker-entrypoint-initdb.d/init.sql
			# 로컬 DB 데이터가 날아가지 않도록 볼륨 마운트
			- ./db_data:/var/lib/mysql
	    healthcheck:
		    # 헬스체크
		    
	# ... 다른 컨테이너 정의
		    
networks:
  frontend-network:
  backend-network:
  data-network:
```

1. YAML Anchor (`&`) 및 Alias (`*`): 공통 설정 재사용
	 - `&` (Anchor): 특정 설정 블록에 이름을 붙여 저장 (예: `&logging-default`)
	- `*` (Alias): 저장된 설정 블록을 그대로 가져와서 사용 (예: `logging: *logging-default`)
2. 환경 변수와 기본값 (.env 파일에 있는 설정값을 읽어옴)
	- **`${VAR:-default}`**: `VAR` 환경 변수가 없으면 `default` 값을 사용
	- **`${VAR:?error}`**: `VAR` 환경 변수가 없으면 에러를 내며 멈춤 (필수 값 체크)
3. `healthcheck`: 서비스 상태 확인
	- `test`: 상태를 확인할 명령어 (예: `mysqladmin ping`, `redis-cli ping`)
	- `interval`: 확인 주기 (예: 30초마다)
	- `timeout`: 응답 대기 시간
	- `retries`: 몇 번 실패해야 'unhealthy'로 간주할지
	- `start_period`: 초기 구동 시 헬스체크를 유예해 줄 시간 (앱 뜨는 시간을 배려)
4. `depends_on`: 실행 순서 제어 (서비스 간의 의존 관계 정의)
	- 기본: 해당 서비스가 시작되면 바로 다음 서비스 실행
	- `condition: service_healthy`: 대상 서비스가 헬스체크를 통과 할 때까지 기다렸다가 실행


### nginx.conf

``` nginx.conf
http {
	# 파일이 무슨 타입(이미지, 실행 파일 등)인지 알려주는 역할
    include       /etc/nginx/mime.types;
    default_type  application/octet-stream;

    sendfile        on;
    keepalive_timeout  65;

    # Nginx 역방향 프록시 (Reverse Proxy) 및 정적 파일 서버 설정
    # -------------------------------------------------------------
    server {
        listen 80;
        server_name localhost;
        
        # 1. Vue.js 빌드 정적 파일 서빙
        # 루트('/') 경로로 들어오는 모든 요청은 Vue 앱을 향합니다.
        location / {
	        # localhost/ 경로와 index.html 파일을 매핑
            root   /usr/share/nginx/html;
            index  index.html;
            # 파일 찾기 우선순위 설정: 1.$uri 2.$uri/ 3.index.html
            # Vue/React는 실제 파일이 index.html 하나뿐이고 index.html이 다른 주소의 화면을 띄워주는 구조임.
            try_files $uri $uri/ /index.html;

            # 헬스체크(wget/curl) 요청 로그 끄기
            if ($http_user_agent ~* "^(curl|wget)") {
                access_log off;
            }
        }

        # 1-1. 정적 파일 (이미지, 아이콘, JS, CSS 등) 장기 캐싱 및 로그 끄기
        location ~* \.(?:ico|css|js|gif|jpe?g|png|svg|woff2?|eot|ttf)$ {
            root   /usr/share/nginx/html;
            expires 6M;          # 6개월 브라우저 캐싱
            access_log off;      # 로그 끄기
            add_header Cache-Control "public";
        }

        # 2. 백엔드 API 서버 프록시 설정
        # '/api/' 로 시작하는 일반 요청 프록시
        location /api/ {
            proxy_pass http://capture-server:8080/;
            
            # 클라이언트의 원래 정보(IP 등)를 백엔드 서버도 알 수 있도록 헤더에 담아서 보냅니다.
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $http_cf_connecting_ip;
            proxy_set_header X-Forwarded-For $http_cf_connecting_ip;
            proxy_set_header X-Forwarded-Proto $scheme;
            
            # 최대 업로드 크기를 제한
            client_max_body_size 500M;
        }
    }
}
```
